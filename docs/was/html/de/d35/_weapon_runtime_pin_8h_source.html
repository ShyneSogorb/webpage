158 template <typename Type, bool DoEnsure = true>
159 void ConvertDataTypeToArrayType() const
160 {
161 // Converts a single stored value into an array payload when the pin is expected to hold arrays.
162 // UObject arrays use strong object pointers and require special handling for object validity.
163 static constexpr bool IsUObject = std::is_base_of_v<UObject, Type>;
164 using T = std::conditional_t<IsUObject, TStrongObjectPtr<Type>, Type>;
165
166 if constexpr (DoEnsure)
167 {
168 if (!EnsureCacheValue())
169 {
170 //Could not get the default value, so try force a default
171 ABORT_RETURN
172 CacheValue->Emplace(TArray<T>());
173 }
174 }
175
176 ABORT_RETURN
177 if(CacheValue->IsArray()) return;
178
179 TArray<T> Arr( { CacheValue->GetTemp<T>() } );
180
181 if constexpr (IsUObject)
182 {
183 UObject* Obj = Arr[0].Get();
184 if (!Obj->IsValidLowLevel())
185 {
186 Arr[0] = nullptr;
187 }
188 UE_LOG(LogWASRuntime, Verbose, TEXT("ConvertDataTypeToArrayType: Converted pin %s data from single value to array type. Element type: %s"), *PinName.ToString(), *GetNameSafe(Arr[0].Get()));
189 }
190 else if constexpr (std::is_arithmetic_v<T>)
191 {
192 UE_LOG(LogWASRuntime, Verbose, TEXT("ConvertDataTypeToArrayType: Converted pin %s data from single value to array type. Element type: %f"), *PinName.ToString(), (double)Arr[0]);
193 }
194 else
195 {
196 UE_LOG(LogWASRuntime, Verbose, TEXT("ConvertDataTypeToArrayType: Converted pin %s data from single value to array type. Element type is not avaible"), *PinName.ToString());
197 }
198
199 CacheValue->Emplace(MoveTemp(Arr));
200 }

204 const TPersistantArray<UObject>* GetObjectArray_Impl() const
205 {
206 ensureMsgf(IsValidType(ObjectSubcategory), TEXT("GetCacheObjectValue: Pin subcategory must be COMPONENT or WILDCARD, but got %s"), *PinSubCategory.ToString());
207
208 if (!EnsureCacheValue())
209 {
210 ABORT_RETURN_VALUE(nullptr)
211 CacheValue->Emplace(TPersistantArray<UObject>());
212 }
213
214 if(!CacheValue->IsArray())
215 {
216 ConvertDataTypeToArrayType<UObject, false>();
217 }
218
219 return CacheValue->Get<TPersistantArray<UObject>>();
220 }

222 template <AnyObjectType T>
223 const TPersistantArray<T>* GetObjectArrayAs_Impl() const
224 {
225 const TPersistantArray<UObject>* Arr = GetObjectArray_Impl();
226 if(!Arr) return nullptr;
227
228#if DO_CHECK
229 for(auto It = Arr->CreateConstIterator(); It; ++It)
230 {
231 if(*It == nullptr) continue; // Allow null entries, but skip the check for them
232 CHECK_ABORT_VALUE(Cast<T>(It->Get()), nullptr,
233 TEXT("GetObjectArrayAs_Impl: Failed to cast object in pin %s to type %s. Object: %s"),
234 *PinName.ToString(), *GetNameSafe(T::StaticClass()), *GetNameSafe(It->Get())
235 )
236 }
237#endif
238 return reinterpret_cast<const TPersistantArray<T>*>(Arr);
239 }

243 template <AnyObjectType T = UObject>
244 FORCEINLINE TStrongObjectPtr<T> GetSingleObjectValueStrongObject(const int32 Index = 0) const
245 {
246 (void)EnsureCacheValue();
247 ABORT_RETURN_VALUE(nullptr)
248 if(!CacheValue->IsValid()) return nullptr;
249
250 if(CacheValue->IsArray())
251 {
252 const auto Arr = GetObjectArray<T>();
253 ABORT_RETURN_VALUE(nullptr)
254 return Arr->IsValidIndex(Index) ? Arr->operator[](Index) : nullptr;
255 }
256 else
257 {
258 if(auto Ptr = CacheValue->GetChecked<TStrongObjectPtr<T>>())
259 return *Ptr;

261 }

319 template <AnyObjectType T>
320 void SetCacheObjectArray(TPersistantArray<T>&& Value) const
321 {
322 ensureMsgf(IsValidType(ObjectSubcategory), TEXT("SetCacheObjectValue: Pin subcategory must be COMPONENT or WILDCARD, but got %s"), *PinSubCategory.ToString());
323 EnsureCachePointer();
324 if constexpr (std::is_same_v<T, UObject>)
325 {
326 CacheValue->Emplace(MoveTemp(Value));
327 }
328 else
329 {
330 // reinterpret_cast because we know T is a UObject-derived type,
331 // and validations above ensure that all elements in Value are valid UObject pointers.
332 TPersistantArray<UObject> Data(
333 reinterpret_cast<TStrongObjectPtr<UObject> const*>(Value.GetData()),
334 Value.Num()
335 );
336
337 Value.Reset(); // Clear the original array to avoid dangling pointers
338
339 CacheValue->Emplace(MoveTemp(Data));
340 }
