24// Compile-time bit position helpers for internal type flags.
25template <size_t N>
26constexpr size_t BitAtPositionLeftStart()
27{
28 static_assert(N < sizeof(size_t) * 8, "N must be less than the number of bits in Type");
29 return size_t(1) << N;
30}
31
32template <size_t N>
33constexpr size_t BitAtPositionRightStart()
34{
35 static_assert(N < sizeof(size_t) * 8, "N must be less than the number of bits in Type");
36
37 constexpr size_t Size = sizeof(size_t) * 8;
38 return size_t(1) << (Size - 1 - N);
39}

41// Wraps compile-time bit shifts for positive and negative movement.
42template <TInt Type, TSInt Movement>
43constexpr Type Move(Movement N, Type Value = 1)
44{
45 if constexpr (N > 0) return MoveLeft(N, Value);
46 if constexpr (N < 0) return MoveRight(N, Value);
47 return Value;
48}

244 template<typename T>
245 void Construct(T&& Value)
246 {
247 static_assert(std::is_unsigned_v<bool> && std::is_integral_v<bool>, "Bool must be consider as unsigned integral for the sake of this struct");
248 if constexpr (std::is_floating_point_v<PureType<T>>)
249 {
250 double val = Value;
251 Construct_Impl<decltype(val)>(Forward<decltype(val)>(val));
252 Type = Double;
253 }
254 else if constexpr (std::is_integral_v<PureType<T>>)
255 {
256 if constexpr (std::is_unsigned_v<PureType<T>>)
257 {
258 uint64 val = Value;
259 Construct_Impl<decltype(val)>(Forward<decltype(val)>(val));
260 Type = UInt64;
261 }
262 else
263 {
264 int64 val = Value;
265 Construct_Impl<decltype(val)>(Forward<decltype(val)>(val));
266 Type = Int64;
267 }
268 }
269 else
270 {
271 Construct_Impl<T>(Forward<T>(Value));
272 Type = Other;
273 }
274 if constexpr (TIsTArray_V<PureType<T>>) //Previous override makes it always false by default
275 {
276 Type |= Array;

280 template <typename T>
281 void Construct_Impl(T&& Value)
282 {
283 using PureT = std::decay_t<T>;
284 Data = new PureT(Forward<T>(Value));
285 Deleter.Lambda = [](void* Ptr)
286 {
287 delete static_cast<PureT*>(Ptr);
288 };
289 }

291 void Destruct()
292 {
293 if (!Data)
294 {
295 check(!Deleter.Lambda)
296 return;
297 }
298 if(IsLambdaDeleted())
299 {
300 check(Deleter.Lambda)
301 Deleter.Lambda(Data);
302 }
303 else if (IsPropertyDeleted())
304 {
305 check(Deleter.Property)
306
307 Deleter.Property->DestroyValue(Data);
308 FMemory::Free(Data);
309 }
310
311 RemoveOwnership();
312 }

321 // Converts a reflected property type into the internal type flag used by FPinValue.
322 static uint8 GetTypeFromProperty(FProperty* Property)
323 {
324 if (auto AsArray = CastField<FArrayProperty>(Property))
325 {
326 return (Array | GetTypeFromProperty(AsArray->Inner));
327 }
328
329 if (auto AsNumeric = CastField<FNumericProperty>(Property))
330 {
331 if (AsNumeric->IsInteger())
332 return IsPropertySigned(AsNumeric) ? Int64 : UInt64;
333 else
334 return Double;
335 }
336
337 return Other;
338 }

341 {
342 check(Deleter.Property);
343 Data = FMemory::Malloc(Deleter.Property->GetSize(), Deleter.Property->GetMinAlignment());
344 Type = GetTypeFromProperty(Deleter.Property) | PropertyDeleter;
345 Deleter.Property->CopyCompleteValue(Data, InData);
