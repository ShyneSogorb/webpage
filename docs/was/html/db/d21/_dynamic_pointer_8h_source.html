17template<typename Comp, typename T, typename U, typename... Ts>
18constexpr auto reduce_types() {
19 using best = std::conditional_t<
20 Comp{}.template operator()<T, U>(),
21 T,
22 U
23 >;
24
25 if constexpr (sizeof...(Ts) == 0) {
26 return type_identity<best>{};
27 } else {
28 return reduce_types<Comp, best, Ts...>();
29 }
30}

623 template <typename List, typename ConvertTo, typename ThisClass, auto Callable, std::size_t... Is, typename... Args>
624 static ConvertTo ExecuteOverType_Impl(ThisClass&& Self, std::index_sequence<Is...>, Args&&... args)
625 {
626
627 constexpr bool IsConst = std::is_const_v<std::remove_reference_t<ThisClass>>;
628 using VoidType = std::conditional_t<IsConst, const void*, void*>;
629
630 using LambdaType = ConvertTo(*)(VoidType, Args&&...);
631
632 static constexpr LambdaType LambdaArray[] = {
633 []<typename... T0>(VoidType Ptr, T0&&... args) {
634 using T = typename List::template ElementAt<Is>;
635 using CurrentType = std::conditional_t<IsConst, std::add_const_t<T>, T>;
636
637 return static_cast<ConvertTo>(Callable(*static_cast<CurrentType*>(Ptr), std::forward<T0>(args)...));
638 }...
639 };
640
641 const auto Index = FMath::CountTrailingZeros(Self.GetSize()) * 2 + (uint8)Self.IsSigned();
642 if(ensureAlways(Index < List::template ListSize))
643 return LambdaArray[Index](Self.GetPtr_Internal(), std::forward<Args>(args)...);
644
645 checkNoEntry(); //This should never happen since we check size and signedness on set, but just in case
646 return ConvertTo{}; //Default value if something goes really wrong, but should never happen
647 }

651 {

670 {

681 template <typename ConvertTo, typename ThisClass, auto Callable, typename ...Args>
682 static ConvertTo ExecuteOverTypeSwitch(ThisClass&& Self, Args&&... args)
683 {
684 check(Self.IsValid())
685 if (Self.IsSigned())
686 {
687 switch (Self.GetSize())
688 {
689 case 1: return Callable(static_cast<BSmallestSigned_t<1>>(Self.GetPtr()), std::forward<Args>(args)...);
690 case 2: return Callable(static_cast<BSmallestSigned_t<2>>(Self.GetPtr()), std::forward<Args>(args)...);
691 case 4: return Callable(static_cast<BSmallestSigned_t<4>>(Self.GetPtr()), std::forward<Args>(args)...);
692 case 8: return Callable(static_cast<BSmallestSigned_t<8>>(Self.GetPtr()), std::forward<Args>(args)...);
693 }
694 }
695 else
696 {
697 switch (Self.GetSize())
698 {
699 case 1: return Callable(static_cast<BSmallestUnsigned_t<1>>(Self.GetPtr()), std::forward<Args>(args)...);
700 case 2: return Callable(static_cast<BSmallestUnsigned_t<2>>(Self.GetPtr()), std::forward<Args>(args)...);
701 case 4: return Callable(static_cast<BSmallestUnsigned_t<4>>(Self.GetPtr()), std::forward<Args>(args)...);
702 case 8: return Callable(static_cast<BSmallestUnsigned_t<8>>(Self.GetPtr()), std::forward<Args>(args)...);
703 }
