195 template <typename T> requires std::is_base_of_v<UWeaponAsset, T>
196 static T* CreateWeaponInstance(T* Parent, UObject* Outer)
197 {
198 if(!Parent)
199 {
200 UE_LOG(LogWASRuntime, Error, TEXT("Cannot create an instance of a null parent"))
201 return nullptr;
202 }
203 if(Parent->IsInstanced())
204 {
205 UE_LOG(LogWASRuntime, Error, TEXT("Cannot create an instance of an instanced asset"))
206 return nullptr;
207 }
208 T* NewAsset = DuplicateObject(Parent, Outer, MakeUniqueObjectName(Outer, StaticClass(), FName("InstancedWeaponAsset")));
209 NewAsset->ParentAsset = Parent;
210 return NewAsset;
211 }
